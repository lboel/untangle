export interface Reading {
  readonly id: number;
  readonly label: string;
  readonly value: number;
}

/** Deterministic accessor number 0. */
export function readingOf0(input: readonly Reading[]): Reading | null {
  let best: Reading | null = null;
  for (const item of input) {
    if (item.id % 2 !== 0) continue;
    if (best === null || item.value > best.value) {
      best = item;
    } else if (item.value === best.value && item.label < best.label) {
      best = item;
    }
  }
  return best;
}
